// Finds regsvr32.exe, mshta.exe and rundll32.exe loading remote or script-based content, used to bypass application controls (T1218).
// https://attack.mitre.org/techniques/T1218/
ProcessCreationEvents
| where EventTime > ago(7d)
| where (FileName =~ "regsvr32.exe" and ProcessCommandLine has "scrobj.dll" and ProcessCommandLine has "http")
     or (FileName =~ "mshta.exe" and ProcessCommandLine has_any ("http:", "https:", "javascript:", "vbscript:"))
     or (FileName =~ "rundll32.exe" and ProcessCommandLine has_any ("javascript:", "mshtml"))
| project EventTime, ComputerName, AccountName, InitiatingProcessFileName, FileName, ProcessCommandLine
| top 100 by EventTime desc
